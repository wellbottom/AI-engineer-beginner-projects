# The shared backend layer is installed as a local *editable* path dependency so
# this service reuses ai_shared (config, errors incl. ImageProviderError/
# ImageTimeoutError, the History_Store models/repository incl. ImageHistory BYTEA +
# ImageGenerationResult/ImageRecord, the best-effort persist_record wrapper, and
# the SSE/error envelope helpers) without duplicating it. Run from this directory:
#   python -m venv .venv
#   .venv\Scripts\python -m pip install -r requirements.txt   (Windows)
#   source .venv/bin/activate && pip install -r requirements.txt  (POSIX)
-e ../../packages/ai_shared

# FastAPI app + ASGI server.
fastapi>=0.115
uvicorn[standard]>=0.30

# Image generation: the Image_Provider is Hugging Face accessed via
# `huggingface_hub` (`InferenceClient.text_to_image`). `huggingface_hub` is already
# a transitive dependency of the editable ai_shared install (it backs
# ai_shared.embeddings.HFEmbeddings), so it is NOT re-declared here; the service
# calls it only through the small lazy-import wrapper app/provider.py
# (HFImageProvider) — mirroring ai_shared's provider-wrapper style (lazy import +
# injectable client) so the validate / encode / timeout / error-mapping logic stays
# importable and unit-testable without a live client or a real HF token.
#
# Pillow converts the PIL image returned by `text_to_image` into raw PNG bytes (the
# durable BYTEA + the web-renderable base64 payload). `huggingface_hub`'s
# text_to_image already requires Pillow at runtime; it is declared explicitly here
# so the byte conversion is guaranteed available.
pillow>=10.0
